33/// @brief Valkey or Redis future for a non-scan and non-eval responses.
34///
35/// Member functions of classes storages::redis::Client and storages::redis::Transaction that do send request to the
36/// Redis return this type or storages::redis::ScanRequest.
37template <typename ResultType, typename ReplyType>
38class [[nodiscard]] Request final {
39public:
40 using Result = ResultType;
41 using Reply = ReplyType;
42
43 explicit Request(std::unique_ptr<RequestDataBase<ReplyType>>&& impl)
44 : impl_(std::move(impl))
45 {}
46
47 /// Wait for the request to finish on Redis server, server or request errors (if any) are logged but not thrown.
48 ///
49 /// @throws Exceptions on misuse (for example, calling Wait() on a single result from a transaction before waiting
50 /// for the transaction itself).
51 void Wait() { impl_->Wait(); }
52
53 /// Ignore the query result and do not wait for the Redis server to finish executing it
54 void IgnoreResult() const noexcept {}
55
56 /// Wait for the request to finish on Redis server and get the result
57 /// @throws server or request related exceptions
58 ReplyType Get(const std::string& request_description = {}) { return impl_->Get(request_description); }
59
60 /// Satisfies @ref engine::Awaitable, for use with @ref engine::WaitAnyContext and friends.
61 engine::AwaitableToken GetAwaitableToken() noexcept USERVER_IMPL_LIFETIME_BOUND {
62 return impl_->GetAwaitableToken();
63 }

66 friend class RequestEval;

75 friend class RequestGeneric;
76
77private:
78 ReplyPtr GetRaw() { return impl_->GetRaw(); }

80 std::unique_ptr<RequestDataBase<ReplyType>> impl_;

111 class Iterator {

127 class ReplyElemHolder {

134
135 private:
136 value_type reply_elem_;
137 };

139 ReplyElemHolder operator++(int) {
140 ReplyElemHolder old_value(stream_->Current());
141 ++*this;

144
