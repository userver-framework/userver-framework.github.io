300def _flatten(query_params):
301 result = []
302 iterable = (
303 query_params.items() if isinstance(query_params, dict) else query_params
304 )
305 for key, value in iterable:
306 if isinstance(value, (tuple, list)):
307 for element in value:
308 result.append((key, str(element)))
309 else:
310 result.append((key, str(value)))
311 return result
