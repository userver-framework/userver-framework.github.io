172TEST_P(Int32FromJsonSuccessTest, Test) {
173 const auto& param = GetParam();
174
175 proto_json::messages::Int32Message message;
176 proto_json::messages::Int32Message expected_message;
177 proto_json::messages::Int32Message sample_message;
178 formats::json::Value input = PrepareJsonTestData(param.input);
179 expected_message = PrepareTestData(param.expected_message);
180
181 message.set_field1(100001);
182 message.set_field2(200002);
183 message.set_field3(300003);
184
185 UASSERT_NO_THROW((message = JsonToMessage<proto_json::messages::Int32Message>(input, param.options)));
186
187#if GOOGLE_PROTOBUF_VERSION >= 6030000
188 UASSERT_NO_THROW(InitSampleMessage(param.input, sample_message, param.options));
189 CheckMessageEqual(message, sample_message);
190#else
191 if (!param.skip_native_parsing_for_older_versions) {
192 UASSERT_NO_THROW(InitSampleMessage(param.input, sample_message, param.options));
193 CheckMessageEqual(message, sample_message);
194 }
195#endif
196
197 CheckMessageEqual(message, expected_message);
198}
