26class ColoredLevelFormatter(logging.Formatter):
27 LEVEL_COLORS = {
28 logging.DEBUG: colors.Colors.GRAY,
29 logging.INFO: colors.Colors.BRIGHT_GREEN,
30 logging.WARNING: colors.Colors.YELLOW,
31 logging.ERROR: colors.Colors.RED,
32 logging.CRITICAL: colors.Colors.BRIGHT_RED,

35 def __init__(self, *, colors_enabled=False):
36 super().__init__()
37 self._colors_enabled = colors_enabled

39 def format(self, record: logging.LogRecord):
40 message = super().format(record)
41 if not self._colors_enabled:
42 return f'{record.levelname} {message}'
43 color = self.LEVEL_COLORS.get(record.levelno, colors.Colors.DEFAULT)
44 return f'{color}{record.levelname}{colors.Colors.DEFAULT} {message}'

145def _command_start(env, args):
146 status = True
147 for service_name in args.services:
148 try:
149 env.ensure_started(service_name)
150 except shell.SubprocessFailed as exc:
151 logger.error(

155 )
156 status = False

188def _register_services(service_plugins=None):
189 services = {}
190
191 def _register_service(name, factory=None):
192 def decorator(factory):
193 services[name] = factory
194
195 if factory is None:
196 return decorator
197 return decorator(factory)
198
199 if service_plugins is None:
200 service_plugins = DEFAULT_SERVICE_PLUGINS
201 strict_check = False
202 else:
203 strict_check = True
204
205 for modname in service_plugins:
206 try:
207 mod = importlib.import_module(modname)
208 except ImportError:
209 if strict_check:
210 raise
211 continue
212 mod.pytest_service_register(register_service=_register_service)
213 return services
