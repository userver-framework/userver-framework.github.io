65Aws::String SqsJsonClient::CreateQueue(sqs::JsonClient& client, std::string_view queue_name) {
66 sqs_model::CreateQueueRequest request;
67 request.SetQueueName(std::string{queue_name}.c_str());
68
69 const auto outcome = client.CreateQueue(request);
70 if (!outcome.IsSuccess()) {
71 ADD_FAILURE() << "Failed to create queue: " << GetErrorMessage(outcome.GetError());
72 return {};
73 }
74
75 return outcome.GetResult().GetQueueUrl();

78void SqsJsonClient::DeleteAllQueues() {
79 auto client = MakeClient();
80 Aws::String next_token;
81
82 do {
83 sqs_model::ListQueuesRequest list_request;
84 if (!next_token.empty()) {
85 list_request.SetNextToken(next_token);
86 }
87
88 const auto list_outcome = client->ListQueues(list_request);
89 if (!list_outcome.IsSuccess()) {
90 ADD_FAILURE() << "Failed to list queues: " << GetErrorMessage(list_outcome.GetError());
91 return;
92 }
93
94 const auto& result = list_outcome.GetResult();
95 for (const auto& queue_url : result.GetQueueUrls()) {
96 sqs_model::DeleteQueueRequest delete_request;
97 delete_request.SetQueueUrl(queue_url);
98
99 const auto delete_outcome = RunWhenQueueIsReady([&] { return client->DeleteQueue(delete_request); });
100 if (!delete_outcome.IsSuccess()) {
101 ADD_FAILURE()
102 << "Failed to delete queue " << queue_url << ": " << GetErrorMessage(delete_outcome.GetError());
103 }
104 }
105
106 next_token = result.GetNextToken();
107 } while (!next_token.empty());
108}
