38// api_url_ может содержать схему и путь (в тестах endpoint указывает на
39// mockserver, например "http://localhost:PORT/s3mds"). Заголовок Host должен
40// нести только authority (host[:port]), поэтому схему и путь отбрасываем.
41// Префикс "bucket." сохраняем всегда: при обращении к localhost GetUrl не
42// кладёт bucket в путь, и mock определяет bucket именно по Host. Внешнее
43// связывание — используется в unit-тесте.
44std::string S3Connection::MakeHostHeader(std::string_view api_url, std::string_view bucket) {
45 const auto schema_pos = api_url.find("://");
46 auto authority = schema_pos == std::string_view::npos ? api_url : api_url.substr(schema_pos + 3);
47
48 const auto path_pos = authority.find('/');
49 if (path_pos != std::string_view::npos) {
50 authority = authority.substr(0, path_pos);
51 }
52
53 if (!bucket.empty()) {
54 return fmt::format("{}.{}", bucket, authority);
55 }
56
57 return std::string{authority};
58}

62std::shared_ptr<clients::http::Response> S3Connection::RequestApi(Request& r, std::string_view method_name) {
63 r.headers[USERVER_NAMESPACE::http::headers::kHost] = GetHostHeader(r);
64 LOG_DEBUG() << "S3 Host: " << r.headers[USERVER_NAMESPACE::http::headers::kHost];
65
66 const std::string full_url = GetUrl(r, connection_type_);
67 LOG_DEBUG() << "S3 file full_url: " << full_url;
68
69 auto http_req =
70 http_client_.CreateNotSignedRequest().timeout(config_.timeout).retry(config_.retries).headers(r.headers);
71
72 if (config_.proxy.has_value()) {
73 http_req.proxy(config_.proxy.value());
74 }
75 http_req
76 .SetDestinationMetricName(fmt::format("{}/{}", r.headers[USERVER_NAMESPACE::http::headers::kHost], method_name)
77 );
78 std::shared_ptr<clients::http::Response> response;
79 try {
80 response = GetMethod(http_req, full_url, r.body, r.method).perform();
81 response->raise_for_status(clients::http::Response::RaiseIncludeBody::kYes);
82 } catch (const clients::http::TimeoutException& e) {
83 LOG_WARNING() << "S3Api : Http Request Timeout: " << full_url;
84 throw;
85 } catch (const clients::http::HttpException& exc) {
86 LOG_INFO() << "S3Api : Http Request to mds failed " << response->body() << " : " << full_url;
87 throw;
88 }
89 return response;
90}

107std::string S3Connection::GetUrl(const Request& r, S3ConnectionType connection_type) const {
108 std::string full_url = api_url_;
109 const bool is_localhost = api_url_.find("localhost:") != std::string::npos;
110 const auto schema_pos = full_url.find("://");
111 if (schema_pos == std::string::npos) {
112 if (!is_localhost && !r.bucket.empty()) {
113 full_url = fmt::format("{}.{}", r.bucket, api_url_);
114 }
115 if (connection_type == S3ConnectionType::kHttps) {
116 full_url = "https://" + full_url;
117 } else {
118 full_url = "http://" + full_url;
119 }
120 } else {
121 if (!is_localhost && !r.bucket.empty()) {
122 const auto schema = full_url.substr(0, schema_pos);
123 const auto schemaless_url = full_url.substr(schema_pos + 3);
124 full_url = fmt::format("{}://{}.{}", schema, r.bucket, schemaless_url);
125 }
126 }
127 if (!r.req.empty()) {
128 full_url += '/';
129 full_url += r.req;
130 }
131 return full_url;
132}
