40/// @brief Event-like helper for waiting until HTTP headers can be sent.
41///
42/// For streamed responses this happens on the first body chunk (or when the
43/// handler finishes). It does not wait for the full response body.
44class HeadersEndEvent final {
45public:
46 explicit HeadersEndEvent(engine::SingleConsumerEvent& event) noexcept : event_(event) {}
47
48 [[nodiscard]] bool IsReady() const noexcept { return event_.IsReady(); }
49
50 /// @brief Satisfies @ref engine::Awaitable, for use with @ref engine::WaitAny.

52 return event_.GetAwaitableToken();
53 }
54
