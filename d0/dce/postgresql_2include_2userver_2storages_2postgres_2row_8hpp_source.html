246template <typename... Args>
247constexpr void AssertSaneTypeToDeserialize() {
248 static_assert(
249 !(IsOptionalFromOptional<std::remove_const_t<std::remove_reference_t<Args>>>::value || ...),
250 "Attempt to get an optional<optional<T>> was detected. Such "
251 "optional-from-optional types are very error prone, obfuscate code and "
252 "are ambiguous to deserialize. Change the type to just optional<T>"

266template <std::size_t... Indexes, typename... T>
267struct RowDataExtractorBase<std::index_sequence<Indexes...>, T...> {
268 static void ExtractValues(const Row& row, T&&... val) {
269 static_assert(sizeof...(Indexes) == sizeof...(T));
270
271 std::size_t field_index = 0;
272 const auto perform = [&](auto&& arg) { row.GetFieldView(field_index++).To(std::forward<decltype(arg)>(arg)); };
273 (perform(std::forward<T>(val)), ...);
274 }
275 static void ExtractTuple(const Row& row, std::tuple<T...>& val) {
276 static_assert(sizeof...(Indexes) == sizeof...(T));
277
278 std::size_t field_index = 0;
279 const auto perform = [&](auto& arg) { row.GetFieldView(field_index++).To(arg); };
280 (perform(std::get<Indexes>(val)), ...);
281 }
282 static void ExtractTuple(const Row& row, std::tuple<T...>&& val) {
283 static_assert(sizeof...(Indexes) == sizeof...(T));
284
285 std::size_t field_index = 0;
286 const auto perform = [&](auto& arg) { row.GetFieldView(field_index++).To(arg); };
287 (perform(std::get<Indexes>(val)), ...);
288 }

327 static_assert(

329 "Row type must be mapped to pg type(CppToUserPg) or one of the "

348template <typename T>
349void Row::To(T&& val, RowTag) const {
350 detail::AssertSaneTypeToDeserialize<T>();
351 // Convert the val into a writable tuple and extract the data
352 using ValueType = std::remove_cvref_t<T>;
353 io::traits::AssertIsValidRowType<ValueType>();
354 using RowType = io::RowType<ValueType>;
355 using TupleType = typename RowType::TupleType;
356 constexpr auto tuple_size = RowType::size;
357 if (tuple_size > Size()) {
358 throw InvalidTupleSizeRequested(Size(), tuple_size);
359 } else if (tuple_size < Size()) {
360 LOG_LIMITED_WARNING()
361 << "Row size is greater that the number of data members in "
362 "C++ user datatype "
363 << compiler::GetTypeName<T>();
364 }
365
366 detail::TupleDataExtractor<TupleType>::ExtractTuple(*this, RowType::GetTuple(std::forward<T>(val)));
367}
368
369template <typename T>
370void Row::To(T&& val, FieldTag) const {
371 detail::AssertSaneTypeToDeserialize<T>();
372 using ValueType = std::remove_cvref_t<T>;
373 detail::AssertRowTypeIsMappedToPgOrIsCompositeType<ValueType>();
374 // Read the first field into the type
375 if (Size() < 1) {
376 throw InvalidTupleSizeRequested{Size(), 1};
377 }
378 if (Size() > 1) {
379 throw NonSingleColumnResultSet{Size(), compiler::GetTypeName<T>(), "As"};
380 }
381 (*this)[0].To(std::forward<T>(val));
382}

384template <typename... T>
385void Row::To(T&&... val) const {
386 detail::AssertSaneTypeToDeserialize<T...>();
387 if (sizeof...(T) > Size()) {
388 throw InvalidTupleSizeRequested(Size(), sizeof...(T));
389 }
390 detail::RowDataExtractor<T...>::ExtractValues(*this, std::forward<T>(val)...);
391}
