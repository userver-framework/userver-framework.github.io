147class _PatchedServicerContext:
148 """
149 `grpc.aio.ServicerContext.abort` immediately sends the response status,
150 before `AsyncCallQueue` has a chance to update its call log.
151 As a workaround, patch `abort` to delay the actual `abort` call until exit from `AsyncCallQueue`.
152 """

154 def __init__(self, context: grpc.aio.ServicerContext) -> None:
155 self._context = context

157 async def abort(self, code: grpc.StatusCode, details: str = '', trailing_metadata: Any = tuple()) -> None:
158 raise _PatchedAbort(code, details, trailing_metadata)

160 async def abort_with_status(self, status: grpc.Status) -> None:
161 await self.abort(status.code, status.details, status.trailing_metadata)

163 def __getattr__(self, item: str) -> Any:
164 return getattr(self._context, item)

173 try:

232def _check_is_valid_response(
233 response: object,
234 method_descriptor: _MethodDescriptor,
235) -> google.protobuf.message.Message:
236 if not isinstance(response, google.protobuf.message.Message):
237 raise ValueError(
238 f'In grpc_mockserver handler for {_get_full_method_name(method_descriptor)}: '

241 )
242 descriptor = type(response).DESCRIPTOR
243 assert isinstance(descriptor, google.protobuf.descriptor.Descriptor)

248 f'got: "{descriptor.full_name}"',

273def _create_servicer_mock(servicer_class: type, /) -> _ServiceMock:
274 _check_is_servicer_class(servicer_class)
275 reflection = _reflect_servicer(servicer_class)
276
277 if reflection:
278 any_method_descriptor: _MethodDescriptor = next(iter(reflection.values()))
279 service_name = any_method_descriptor.containing_service.full_name
280 assert '/' not in service_name, service_name
281 else:
282 service_name = 'UNKNOWN'
283
284 adder = getattr(inspect.getmodule(servicer_class), f'add_{servicer_class.__name__}_to_server')
285
286 state = _ServiceMockState()
287
288 methods = {}
289 for attname, value in servicer_class.__dict__.items():
290 if callable(value) and not attname.startswith('_'):
291 methods[attname] = _wrap_grpc_method(
292 python_method_name=attname,
293 method_descriptor=reflection[attname],
294 default_unimplemented_method=value,
295 state=state,
296 )
297 mocked_servicer_class = type(
298 f'Mock{servicer_class.__name__}',
299 (servicer_class,),
300 methods,
301 )
302 servicer = mocked_servicer_class()
303
304 return _ServiceMock(
305 servicer=servicer,
306 adder=adder,
307 service_name=service_name,
308 known_methods=frozenset(methods),
309 state=state,
310 )
