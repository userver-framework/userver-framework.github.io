179UTEST_F(SqsJsonClient, SendReceiveMessageAttributes) {
180 auto client = MakeClient();
181 const auto queue_url = CreateQueue(*client, "json-client-attrs");
182 ASSERT_FALSE(queue_url.empty());
183
184 sqs_model::SendMessageRequest send_request;
185 send_request.SetQueueUrl(queue_url);
186 send_request.SetMessageBody("message-with-attributes");
187 send_request.AddMessageAttributes(
188 "stringAttr",
189 sqs_model::MessageAttributeValue{}.WithDataType("String").WithStringValue("string-value")
190 );
191
192 const unsigned char binary_value[] = {'b', 'i', 'n'};
193 send_request.AddMessageAttributes(
194 "binaryAttr",
195 sqs_model::MessageAttributeValue{}
196 .WithDataType("Binary")
197 .WithBinaryValue(Aws::Utils::ByteBuffer{binary_value, sizeof(binary_value)})
198 );
199
200 const auto send_outcome = RunWhenQueueIsReady([&] { return client->SendMessage(send_request); });
201 ASSERT_TRUE(send_outcome.IsSuccess()) << GetErrorMessage(send_outcome.GetError());
202
203 sqs_model::ReceiveMessageRequest receive_request;
204 receive_request.SetQueueUrl(queue_url);
205 receive_request.SetMaxNumberOfMessages(1);
206 receive_request.SetWaitTimeSeconds(5);
207 receive_request.AddMessageAttributeNames("All");
208
209 const auto receive_outcome = client->ReceiveMessage(receive_request);
210 ASSERT_TRUE(receive_outcome.IsSuccess()) << GetErrorMessage(receive_outcome.GetError());
211 ASSERT_EQ(receive_outcome.GetResult().GetMessages().size(), 1);
212
213 const auto& attributes = receive_outcome.GetResult().GetMessages()[0].GetMessageAttributes();
214 EXPECT_EQ(attributes.at("stringAttr").GetStringValue(), "string-value");
215 EXPECT_EQ(attributes.at("binaryAttr").GetBinaryValue().GetLength(), sizeof(binary_value));
216}
