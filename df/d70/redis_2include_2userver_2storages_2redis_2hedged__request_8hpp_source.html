37template <typename RedisRequestType>
38struct RedisRequestStrategy {
39 using RequestType = RedisRequestType;
40 using ReplyType = typename RequestType::Reply;
41 using GenF = std::function<std::optional<RequestType>(int)>;
42
43 explicit RedisRequestStrategy(GenF gen_callback)
44 : gen_callback_(std::move(gen_callback))
45 {}
46
47 RedisRequestStrategy(RedisRequestStrategy&& other) noexcept = default;
48
49 /// @{
50 /// Methods needed by HedgingStrategy
51 std::optional<RequestType> Create(size_t try_count) { return gen_callback_(try_count); }
52 std::optional<std::chrono::milliseconds> ProcessReply(RequestType&& request) {
53 reply_ = std::move(request).Get();
54 return std::nullopt;
55 }
56 std::optional<ReplyType> ExtractReply() { return std::move(reply_); }
57 void Finish(RequestType&&) {}
58 /// @}
59
60private:
61 GenF gen_callback_;
62 std::optional<ReplyType> reply_;
63};

70template <

80) {

121/// Same as MakeHedgedRedisRequest but accepting a vector of argument tuples
122/// instead of a single set of arguments. And return vector of replies
123/// corresponding to input args.
124template <
125 typename RedisRequestType,
126 typename... Args,
127 typename M = RedisRequestType (storages::redis::Client::*)(Args..., const redis::CommandControl&)>
128std::vector<std::optional<typename RedisRequestType::Reply>> MakeBulkHedgedRedisRequest(
129 std::shared_ptr<storages::redis::Client> redis,
130 M method,
131 const redis::CommandControl& cc,
132 utils::hedging::HedgingSettings hedging_settings,
133 std::vector<std::tuple<Args...>> args
134) {
135 std::vector<redis::impl::RedisRequestStrategy<RedisRequestType>> strategies;
136 strategies.reserve(args.size());
137
138 for (auto&& arg : args) {
139 auto gen_request = [redis, method, cc{cc}, args_tuple{std::move(arg)}](int try_count) mutable
140 -> std::optional<RedisRequestType> {
141 cc.retry_counter = try_count;
142 cc.max_retries = 1; ///< We do retries ourselves
143
144 return std::apply(
145 [redis, method, cc](auto&&... args) { return (redis.get()->*method)(args..., cc); },
146 args_tuple
147 );
148 };
149 strategies.emplace_back(std::move(gen_request));
150 }
151
152 return utils::hedging::HedgeRequestsBulk(std::move(strategies), hedging_settings);
153}
