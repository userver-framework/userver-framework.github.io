172TEST_P(RepeatedFromJsonSuccessTest, Test) {
173 using Message = proto_json::messages::RepeatedMessage;
174 const auto& param = GetParam();
175
176 Message message;
177 Message expected_message;
178 Message sample_message;
179 formats::json::Value input = PrepareJsonTestData(param.input);
180 expected_message = PrepareTestData(param.expected_message);
181
182 UASSERT_NO_THROW((message = JsonToMessage<Message>(input, param.options)));
183 UASSERT_NO_THROW(InitSampleMessage(param.input, sample_message, param.options));
184
185 CheckMessageEqual(message, sample_message);
186 CheckMessageEqual(message, expected_message);
187}
188
189TEST_P(RepeatedFromJsonFailureTest, Test) {
190 using Message = proto_json::messages::RepeatedMessage;
191 const auto& param = GetParam();
192
193 Message sample;
194 formats::json::Value input = PrepareJsonTestData(param.input);
195
196 EXPECT_PARSE_ERROR((void)JsonToMessage<Message>(input, param.options), param.expected_errc, param.expected_path);
197
198 if (!param.skip_native_check) {
199 UEXPECT_THROW(InitSampleMessage(param.input, sample, param.options), SampleError);
200 }
201}
