20/// Status of one input row in an ODBC bulk execution.
21enum class BulkRowStatus {
22 /// Driver confirmed successful execution without diagnostics.
23 kSuccess,
24 /// Driver confirmed success and reported warning-class diagnostics.
25 kSuccessWithInfo,
26 /// Driver reported that this row failed.
27 kError,
28 /// Driver reported that this row was not used.
29 kUnused,
30 /// The row was processed, but row-specific diagnostics are unavailable.
31 kDiagnosticsUnavailable,
32 /// The driver did not provide a trustworthy per-row status.
33 kUnknown,
34};

36/// @brief Owning, ordered rows of parameters for ODBC bulk DML.

41/// untyped and are rejected by bulk preflight.

46 BulkParameterStore(BulkParameterStore&&) noexcept = default;

76/// @brief Outcome snapshot of an ODBC bulk execution.
77///
78/// `Processed()` is absent when the driver did not provide a reliable count.
79/// `RowsAffected()` is absent when any DML result reported an unknown count.
80/// `Succeeded()` counts only kSuccess and kSuccessWithInfo rows; unknown and
81/// diagnostics-unavailable rows are deliberately not assumed successful.
82class BulkResult final {
83public:
84 BulkResult() = default;
85 BulkResult(
86 std::size_t requested,
87 std::optional<std::size_t> processed,
88 std::optional<std::size_t> rows_affected,
89 std::vector<BulkRowStatus> statuses
90 );
91
92 /// Number of input rows. Always equals `Statuses().size()`.
93 std::size_t Requested() const noexcept { return requested_; }
94 /// Reliable processed-row count, if supplied by the driver.
95 std::optional<std::size_t> Processed() const noexcept { return processed_; }
96 /// Number of rows with a confirmed successful status.
97 std::size_t Succeeded() const noexcept { return succeeded_; }
98 /// Checked aggregate DML row count, or null when any count is unknown.
99 std::optional<std::size_t> RowsAffected() const noexcept { return rows_affected_; }
100 /// One status for every requested row, including unused tail rows.
101 const std::vector<BulkRowStatus>& Statuses() const noexcept { return statuses_; }
102
103private:
104 std::size_t requested_{0};
105 std::optional<std::size_t> processed_{0};
106 std::size_t succeeded_{0};
107 std::optional<std::size_t> rows_affected_{0};
108 std::vector<BulkRowStatus> statuses_;
109};
