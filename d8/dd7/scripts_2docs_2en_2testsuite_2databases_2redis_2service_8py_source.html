54class ClusterServiceSettings(typing.NamedTuple):
55 host: str
56 cluster_ports: tuple[int, ...]
57 cluster_replicas: int
58
59 def validate(self):
60 if len(self.cluster_ports) % (self.cluster_replicas + 1) != 0:
61 raise NotEnoughPorts(
62 f'Number of nodes does not match number of replicas ({self.cluster_replicas})',
63 )
64 min_masters = (self.cluster_replicas + 1) * 3
65 if len(self.cluster_ports) < min_masters:
66 raise NotEnoughPorts(
67 f'Need at least {min_masters} cluster nodes!',
68 )

118def create_redis_service(
119 service_name,
120 working_dir,
121 settings: ServiceSettings | None = None,
122 env=None,
123):
124 if settings is None:
125 settings = get_service_settings()
126 configs_dir = pathlib.Path(working_dir).joinpath('configs')
127 check_ports = [
128 settings.sentinel_port,
129 *settings.master_ports,
130 *settings.slave_ports,
131 ]
132
133 def prestart_hook():
134 configs_dir.mkdir(parents=True, exist_ok=True)
135 settings.validate()
136 genredis.generate_redis_configs(
137 output_path=configs_dir,
138 host=settings.host,
139 master0_port=settings.master_ports[0],
140 master1_port=settings.master_ports[1],
141 slave0_port=settings.slave_ports[0],
142 slave1_port=settings.slave_ports[1],
143 slave2_port=settings.slave_ports[2],
144 sentinel_port=settings.sentinel_port,
145 )
146
147 return service.ScriptService(
148 service_name=service_name,
149 script_path=str(SERVICE_SCRIPT_PATH),
150 working_dir=working_dir,
151 environment={
152 'REDIS_TMPDIR': working_dir,
153 'REDIS_CONFIGS_DIR': str(configs_dir),
154 'REDIS_SENTINEL_PORT': str(settings.sentinel_port),
155 'REDIS_SENTINEL_HOST': settings.host,
156 **(env or {}),
157 },
158 check_host=settings.host,
159 check_ports=check_ports,

238def _get_hostname():
239 hostname = 'localhost'
240 for var in ('TESTSUITE_REDIS_HOSTNAME', 'HOSTNAME'):
241 if var in os.environ:
242 hostname = os.environ[var]
243 break
244 return _resolve_hostname(hostname)

247def _resolve_hostname(hostname: str) -> str:
248 for family in socket.AF_INET6, socket.AF_INET:
249 try:
250 result = socket.getaddrinfo(
251 hostname,
252 None,
253 family=family,
254 type=socket.SOCK_STREAM,
255 )
256 except OSError:
257 continue
258 if result:
259 return result[0][4][0] # type: ignore[return-value]
260 warnings.warn(f'Failed to resolve hostname {hostname}')
261 return hostname
