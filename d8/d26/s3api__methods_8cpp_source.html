140 params.emplace(query_args::kDelimiter, delimiter);
141 }
142
143 req.req = "?" + USERVER_NAMESPACE::http::MakeQuery(params);
144 return req;

207Request CreateInternalApiRequest(
208 const std::string& bucket,
209 const multipart_upload::CompleteMultipartUploadRequest& request
210) {
211 Request result;
212 result.method = clients::http::HttpMethod::kPost;
213 result.bucket = bucket;
214 result.req = fmt::format(
215 "{}?{}",
216 http::EncodeS3Key(request.key),
217 http::MakeQuery({{query_args::kUploadId, request.upload_id}})
218 );
219
220 pugi::xml_document doc;
221 auto multipart_upload_node = doc.append_child("CompleteMultipartUpload");
222 multipart_upload_node.append_attribute("xmlns").set_value(kS3AwsXmlNamespace.c_str());
223 for (const auto& part : request.completed_parts) {
224 auto part_node = multipart_upload_node.append_child("Part");
225 part_node.append_child("PartNumber")
226 .append_child(pugi::node_pcdata)
227 .set_value(std::to_string(part.part_number).c_str());
228 part_node.append_child("ETag").append_child(pugi::node_pcdata).set_value(part.etag.c_str());
229 }
230 StringWriter writer{result.body};
231 doc.save(writer, "", pugi::format_raw);
232 result.headers = clients::http::Headers{
233 {http::headers::kContentType, content_types::kApplicationXml},
234 {http::headers::kContentLength, std::to_string(result.body.size())}
235 };
236
237 return result;
238}

240Request CreateInternalApiRequest(const std::string& bucket, const multipart_upload::UploadPartRequest& request) {
241 Request result;
242 result.method = clients::http::HttpMethod::kPut;
243 // `Content-Type`-header is absent in S3 API, however PUT-request implementation in CURL for URIs with query
244 // may implicitly insert `application/x-www-form-urlencoded` which must be taken into account when making
245 // canonical request for signature. But it happens after we sign the request without knowledge of this header.
246 // We explicitly override this Content-Type value here to make this request be signed correctly.
247 result.headers = clients::http::Headers{
248 {http::headers::kContentLength, std::to_string(request.data.size())},
249 {http::headers::kContentType, std::string{content_types::kDefault}}
250 };
251
252 result.bucket = bucket;
253 result.req = fmt::format(
254 "{}?{}",
255 http::EncodeS3Key(request.key),
256 http::MakeQuery(
257 {{query_args::kUploadId, request.upload_id}, {query_args::kPartNumber, std::to_string(request.part_number)}}
258 )
259 );
260 result.body = std::move(request.data);
261 return result;
262}

264Request CreateInternalApiRequest(const std::string& bucket, const multipart_upload::ListPartsRequest& request) {
265 Request result;
266 result.method = clients::http::HttpMethod::kGet;
267 result.bucket = bucket;
268 result.req = http::EncodeS3Key(request.key);
269
270 http::Args params{{std::string{query_args::kUploadId}, request.upload_id}};
271
272 if (request.max_parts > 0 && request.max_parts < kMaxUploadsListLimit) {
273 params.emplace(query_args::kMaxParts, std::to_string(request.max_parts));
274 }
275
276 if (auto part_number_marker = request.part_number_marker.value_or(0u)) {
277 params.emplace(query_args::kPartNumberMarker, std::to_string(part_number_marker));
278 }
279
280 result.req = fmt::format("{}?{}", http::EncodeS3Key(request.key), http::MakeQuery(params));
281
282 return result;
283}

285Request CreateInternalApiRequest(
286 const std::string& bucket,
287 const multipart_upload::ListMultipartUploadsRequest& request
288) {
289 using EncodingType = multipart_upload::ListMultipartUploadsRequest::EncodingType;
290
291 Request result;
292 result.method = clients::http::HttpMethod::kGet;
293 result.bucket = bucket;
294
295 http::Args params;
296 if (request.delimiter) {
297 params.emplace(query_args::kDelimiter, *request.delimiter);
298 }
299
300 if (request.prefix) {
301 params.emplace(query_args::kPrefix, *request.prefix);
302 }
303
304 if (request.max_uploads > 0 && request.max_uploads < kMaxUploadsListLimit) {
305 params.emplace(query_args::kMaxUploads, std::to_string(request.max_uploads));
306 }
307
308 if (request.key_marker) {
309 params.emplace(query_args::kMarker, *request.key_marker);
310 }
311
312 if (request.upload_id_marker) {
313 params.emplace(query_args::kUploadIdMarker, *request.upload_id_marker);
314 }
315
316 if (request.encoding_type == EncodingType::kUrl) {
317 params.emplace(query_args::kEncodingType, kEncodingTypeValueUrl);
318 }
319
320 result.req = "?uploads";
321 if (!params.empty()) {
322 result.req += "&" + http::MakeQuery(params);
323 }
324
325 return result;
326}
