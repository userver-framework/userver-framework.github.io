149def _get_port(
150 original_service_config,
151 choose_free_port,
152 listener_name,
153 port_fixture,
154 option_name,
155) -> int:
156 config_yaml = original_service_config.config_yaml
157 config_vars = original_service_config.config_vars
158 components = config_yaml['components_manager']['components']
159 listener = components.get('server', {}).get(listener_name, {})
160 if not listener:
161 return -1
162
163 port = listener.get('port', None) or listener.get('ports', [None])[0]
164 if isinstance(port, str) and port.startswith('$'):
165 port = config_vars.get(port[1:], None) or listener.get(
166 'port#fallback',
167 None,
168 )
169 assert port, (
170 f'Please specify '
171 f'components_manager.components.server.{listener_name}.port '
172 f'in the static config, or pass {option_name} pytest option, '
173 f'or override the {port_fixture.__name__} fixture'
174 )
175 return choose_free_port(port)

218def _is_port_free(port_num: int, family: int, address: str) -> bool:
219 try:
220 with socket.socket(family, socket.SOCK_STREAM) as sock:
221 sock.bind((address, port_num))
222 except OSError:
223 return False
224 else:
225 return True
