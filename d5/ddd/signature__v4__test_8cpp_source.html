45TEST(S3ApiSignatureV4, AuthGetObject) {
46 auto request = MakeRequest(clients::http::HttpMethod::kGet, "test.txt");
47 request.headers[USERVER_NAMESPACE::http::headers::kRange] = "bytes=0-9";
48
49 const auto headers = MakeAuthenticator().Auth(request);
50
51 EXPECT_EQ(headers.at("X-Amz-Date"), "20130524T000000Z");
52 EXPECT_EQ(headers.at("X-Amz-Content-Sha256"), kEmptyPayloadHash);
53 EXPECT_EQ(
54 headers.at("Authorization"),
55 "AWS4-HMAC-SHA256 Credential=AKIAIOSFODNN7EXAMPLE/20130524/us-east-1/s3/aws4_request, "
56 "SignedHeaders=host;range;x-amz-content-sha256;x-amz-date, "
57 "Signature=f0e8bdb87c964420e857bd35b5d6ed310bd44f0170aba48dd91039c6036bdb41"
58 );
59}
