102TEST_P(UInt64FromJsonSuccessTest, Test) {
103 const auto& param = GetParam();
104
105 proto_json::messages::UInt64Message message;
106 proto_json::messages::UInt64Message expected_message;
107 proto_json::messages::UInt64Message sample_message;
108 formats::json::Value input = PrepareJsonTestData(param.input);
109 expected_message = PrepareTestData(param.expected_message);
110
111 message.set_field1(100001);
112 message.set_field2(200002);
113
114 UASSERT_NO_THROW((message = JsonToMessage<proto_json::messages::UInt64Message>(input, param.options)));
115
116#if GOOGLE_PROTOBUF_VERSION >= 6030000
117 UASSERT_NO_THROW(InitSampleMessage(param.input, sample_message, param.options));
118 CheckMessageEqual(message, sample_message);
119#else
120 if (!param.skip_native_parsing_for_older_versions) {
121 UASSERT_NO_THROW(InitSampleMessage(param.input, sample_message, param.options));
122 CheckMessageEqual(message, sample_message);
123 }
124#endif
125
126 CheckMessageEqual(message, expected_message);
127}
